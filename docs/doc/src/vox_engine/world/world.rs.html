1use super::{Generator, helpers, noise::FractalNoise};
2use getset::CloneGetters;
3use rapier3d::glamx::{IVec3, Vec3};
4use std::collections::HashMap;
5
6#[derive(CloneGetters)]
7pub struct World {
8    #[getset(get_clone = "pub with_prefix")]
9    pub(super) world: Vec<u128>,
10    #[getset(get_clone = "pub with_prefix")]
11    pub(super) dimensions: Vec3,
12    pub(super) chunked_coords: HashMap<IVec3, Vec<IVec3>>,
13    #[getset(get_clone = "pub with_prefix")]
14    pub(super) biomes: Vec<u8>,
15    pub(super) collider_chunk_size: i32,
16}
17
18impl World {
19    pub fn new(
20        seed: u64,
21        dimensions: Vec3,
22        collider_chunk_size: i32,
23        _generator: Option<Box<dyn Generator>>,
24    ) -> Self {
25        let height_map = FractalNoise::new(seed, 5, 2.0, 0.5);
26        let temperature_map = FractalNoise::new((seed as i64 - 1).abs() as u64, 5, 2.0, 0.5);
27        let moisture_map = FractalNoise::new((seed as i64 + 1).abs() as u64, 5, 2.0, 0.5);
28
29        let texel_x = dimensions.x as usize / 4;
30        let texel_y = dimensions.y as usize / 4;
31        let texel_z = dimensions.z as usize / 8;
32
33        let total = texel_x * texel_y * texel_z;
34
35        let mut world = vec![0u128; total];
36        let mut biomes = vec![0u8; ((dimensions.x * dimensions.z) * 4.0) as usize];
37
38        let mut set_voxel = |x: usize, y: usize, z: usize| {
39            let (texel, bit) = helpers::voxel_texel_and_bit(
40                x as i32,
41                y as i32,
42                z as i32,
43                texel_x as i32,
44                texel_y as i32,
45            );
46
47            world[texel] |= 1u128 << bit;
48        };
49
50        let mut set_biome =
51            |x: usize, y: usize, x_len: usize, step: usize, offset: usize, value: u8| {
52                let index = (x * x_len) + y;
53
54                biomes[(index * step) + offset] = value;
55            };
56
57        let noise_frequency = 1.0 / 450.0;
58        let temperature_moisture_freq = 1.0 / 800.0;
59        for x in 0..(dimensions.x as usize) {
60            for z in 0..(dimensions.z as usize) {
61                let height_factor =
62                    height_map.sample(x as f64 * noise_frequency, z as f64 * noise_frequency);
63                let temperature_raw = temperature_map.sample(
64                    x as f64 * temperature_moisture_freq,
65                    z as f64 * temperature_moisture_freq,
66                );
67                let moisture_raw = moisture_map.sample(
68                    x as f64 * temperature_moisture_freq,
69                    z as f64 * temperature_moisture_freq,
70                );
71
72                let temperature = ((temperature_raw * 0.5 + 0.5) * 255.0) as u8;
73                let moisture = ((moisture_raw * 0.5 + 0.5) * 255.0) as u8;
74                let height_factor = height_factor * 0.5 + 0.5;
75
76                set_biome(x, z, dimensions.x as usize, 4, 0, temperature as u8);
77                set_biome(
78                    x,
79                    z,
80                    dimensions.x as usize,
81                    4,
82                    1,
83                    (height_factor * 255.0) as u8,
84                );
85                set_biome(x, z, dimensions.x as usize, 4, 2, moisture as u8);
86                set_biome(x, z, dimensions.x as usize, 4, 3, 255);
87
88                let height = (dimensions.y as f64 * height_factor) as usize;
89                let height = height.clamp(1, dimensions.y as usize);
90                for y in 0..height {
91                    set_voxel(x, y, z);
92                }
93            }
94        }
95
96        let mut world_struct = World {
97            world,
98            dimensions,
99            chunked_coords: HashMap::new(),
100            biomes,
101            collider_chunk_size,
102        };
103        world_struct.chunked_coords = world_struct.compute_chunked_surface_coords();
104        world_struct
105    }
106
107    pub fn get_dimensions_as_arr(&self) -> [u32; 3] {
108        [
109            self.dimensions.x as u32,
110            self.dimensions.y as u32,
111            self.dimensions.z as u32,
112        ]
113    }
114
115    pub fn get_world_as_u32(&self) -> Vec<u32> {
116        self.world
117            .iter()
118            .flat_map(|v| {
119                let bytes = v.to_le_bytes();
120                [
121                    u32::from_le_bytes(bytes[0..4].try_into().unwrap()),
122                    u32::from_le_bytes(bytes[4..8].try_into().unwrap()),
123                    u32::from_le_bytes(bytes[8..12].try_into().unwrap()),
124                    u32::from_le_bytes(bytes[12..16].try_into().unwrap()),
125                ]
126            })
127            .collect()
128    }
