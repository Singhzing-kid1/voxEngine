35impl Ui {
36    pub fn new(engine: &mut Engine) -> Self {
37        let elements = Vec::new();
38        let textures = Atlas::new(Format::R8G8B8A8_UNORM, 0f32, engine);
39
40        let vertex_shader = vertex::load(engine.get_device()).unwrap();
41        let fragment_shader = fragment::load(engine.get_device()).unwrap();
42
43        let pipeline = helpers::create_graphics_pipeline(
44            vertex_shader,
45            fragment_shader,
46            "main",
47            textures.get_format(),
48            engine.get_device(),
49        );
50
51        Self {
52            elements,
53            textures,
54            pipeline,
55
56            vertex_buffer: None,
57            index_buffer: None,
58        }
59    }
60
61    pub fn add_element(&mut self, engine: &mut Engine, element: Box<dyn Element>) {
62        self.textures.add_image(
63            element.get_texture().clone(),
64            engine,
65            true,
66            Some(helpers::get_layout(self.pipeline.layout(), 0)),
67        );
68
69        self.elements.push(element);
70    }
71}

73impl Record for Ui {
74    fn init(&mut self, engine: &mut Engine) {
75        let mut vertices: Vec<UiVertex> = Vec::new();
76        let mut indices: Vec<u32> = Vec::new();
77
78        for e in self.elements.iter_mut() {
79            let base = vertices.len() as u32;
80            e.set_first_idx(indices.len() as u32);
81
82            let (verts, inds, _) = e.get_tris();
83
84            vertices.extend([
85                UiVertex {
86                    pos: verts[0].to_array(),
87                    color: u8vec4(255, 255, 255, 255).to_array(),
88                },
89                UiVertex {
90                    pos: verts[1].to_array(),
91                    color: u8vec4(255, 255, 255, 255).to_array(),
92                },
93                UiVertex {
94                    pos: verts[2].to_array(),
95                    color: u8vec4(255, 255, 255, 255).to_array(),
96                },
97                UiVertex {
98                    pos: verts[3].to_array(),
99                    color: u8vec4(255, 255, 255, 255).to_array(),
100                },
101            ]);
102            indices.extend([base, base + 1, base + 2, base, base + 2, base + 3]);
103        }
104
105        let vertex_buffer = Buffer::from_iter(
106            engine.get_memory_allocator(),
107            BufferCreateInfo {
108                usage: BufferUsage::VERTEX_BUFFER,
109                ..Default::default()
110            },
111            AllocationCreateInfo {
112                memory_type_filter: MemoryTypeFilter::PREFER_DEVICE
113                    | MemoryTypeFilter::HOST_SEQUENTIAL_WRITE,
114                ..Default::default()
115            },
116            vertices,
117        )
118        .unwrap();
119
120        let index_buffer = Buffer::from_iter(
121            engine.get_memory_allocator(),
122            BufferCreateInfo {
123                usage: BufferUsage::INDEX_BUFFER,
124                ..Default::default()
125            },
126            AllocationCreateInfo {
127                memory_type_filter: MemoryTypeFilter::PREFER_DEVICE
128                    | MemoryTypeFilter::HOST_SEQUENTIAL_WRITE,
129                ..Default::default()
130            },
131            indices,
132        )
133        .unwrap();
134
135        self.vertex_buffer = Some(vertex_buffer);
136        self.index_buffer = Some(index_buffer);
137    }
138
139    fn record(&mut self, frame: &mut Frame, engine: &mut Engine) {
140        let screen_size = [
141            engine.get_dimensions().0 as f32,
142            engine.get_dimensions().1 as f32,
143        ];
144
145        let push_data = vertex::PC { screen_size };
146
147        let builder = frame.get_builder_mut();
148
149        builder
150            .set_viewport(
151                0,
152                [Viewport {
153                    offset: [0.0, 0.0],
154                    extent: screen_size,
155                    depth_range: 0.0..=1.0,
156                }]
157                .into_iter()
158                .collect(),
159            )
160            .unwrap()
161            .set_scissor(
162                0,
163                [Scissor {
164                    offset: [0, 0],
165                    extent: [
166                        engine.get_dimensions().0 as u32,
167                        engine.get_dimensions().1 as u32,
168                    ],
169                }]
170                .into_iter()
171                .collect(),
172            )
173            .unwrap()
174            .bind_pipeline_graphics(self.pipeline.clone())
175            .unwrap()
176            .push_constants(self.pipeline.layout().clone(), 0, push_data)
177            .unwrap()
178            .bind_vertex_buffers(0, self.vertex_buffer.as_ref().unwrap().clone())
179            .unwrap()
180            .bind_index_buffer(self.index_buffer.as_ref().unwrap().clone())
181            .unwrap();
182
183        for e in self.elements.iter() {
184            builder
185                .bind_descriptor_sets(
186                    PipelineBindPoint::Graphics,
187                    self.pipeline.layout().clone(),
188                    0,
189                    self.textures
190                        .get_texture(e.get_texture().to_str().unwrap().to_string())
191                        .1
192                        .unwrap()
193                        .clone(),
194                )
195                .unwrap();
196
197            let (_, _, first_index) = e.get_tris();
198
199            unsafe {
200                builder.draw_indexed(6, 1, first_index, 0, 0).unwrap();
201            }
202        }
203    }
