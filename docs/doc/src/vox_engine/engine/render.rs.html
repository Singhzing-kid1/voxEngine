3use vulkano::{
4    command_buffer::{AutoCommandBufferBuilder, CommandBufferUsage},

9impl Engine {
10    pub fn start_frame(&mut self) -> Frame {
11        let (image_index, _, acquire_future) =
12            acquire_next_image(self.swapchain.clone(), None).unwrap();
13
14        self.current_image_index = image_index;
15        let swapchain_image = self.images[image_index as usize].clone();
16
17        let builder = AutoCommandBufferBuilder::primary(
18            self.command_buffer_allocator.clone(),
19            self.queue.queue_family_index(),
20            CommandBufferUsage::OneTimeSubmit,
21        )
22        .unwrap();
23
24        Frame::new(builder, swapchain_image, acquire_future)
25    }
26
27    pub fn finish_frame(&mut self, frame: Frame) {
28        let command_buffer = frame.builder.build().unwrap();
29
30        let previous_future = self
31            .previous_future
32            .take()
33            .unwrap_or_else(|| Box::new(now(self.device.clone())));
34
35        let future = previous_future
36            .join(frame.acquire_future)
37            .then_execute(self.queue.clone(), command_buffer)
38            .unwrap()
39            .boxed_send_sync();
40
41        self.previous_future = Some(Box::new(future));
42
43        if let Some(f) = &mut self.previous_future {
44            f.flush().unwrap();
45        }
46    }
47
48    pub fn present(&mut self) {
49        let future = self.previous_future.take().unwrap();
50
51        let present = future
52            .then_swapchain_present(
53                self.queue.clone(),
54                SwapchainPresentInfo::swapchain_image_index(
55                    self.swapchain.clone(),
56                    self.current_image_index,
57                ),
58            )
59            .then_signal_fence_and_flush()
60            .unwrap();
61
62        present.wait(None).unwrap();
63
64        self.previous_future = Some(Box::new(present));
65
66        if let Some(prev) = &mut self.previous_future {
67            prev.as_mut().cleanup_finished();
68        }
69    }
70}
