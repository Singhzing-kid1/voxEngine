56    pub fn add_to_pitch(&mut self, value: f32) {
57        self.pitch += value;
58
59        if self.pitch > 89.9 {

66    pub fn get_pixel_to_ray_matrix(&self, render_scale: u16) -> Mat4 {
67        let (w, h) = self.dimensions;
68        let (w, h) = ((w / render_scale) as f32, (h / render_scale) as f32);
69
70        let tan_fov = (self.fov.to_radians() * 0.5).tan();
71
72        let f_y = (h * 0.5) / tan_fov;
73        let f_x = f_y;
74
75        let cx = w * 0.5;
76        let cy = h * 0.5;
77
78        let k_inv = Mat3::from_cols(
79            vec3(1.0 / f_x, 0.0, 0.0),
80            vec3(0.0, -1.0 / f_y, 0.0),
81            vec3(-cx / f_x, cy / f_y, -1.0),
82        );
83
84        let r = Mat3::from_quat(self.camera_orientation);
85
86        let m = r * k_inv;
87
88        Mat4::from_cols(
89            vec4(m.col(0).x, m.col(0).y, m.col(0).z, 0.0),
90            vec4(m.col(1).x, m.col(1).y, m.col(1).z, 0.0),
91            vec4(m.col(2).x, m.col(2).y, m.col(2).z, 0.0),
92            vec4(
93                self.camera_position.x,
94                self.camera_position.y,
95                self.camera_position.z,
96                1.0,
97            ),
98        )
99    }
100}

102impl Camera {
103    fn update_orientation(&mut self) {
104        let r_yaw = self.yaw.to_radians();
105        let r_pitch = self.pitch.to_radians();
106
107        let q_yaw = Quat::from_axis_angle(Vec3::Y, r_yaw);
108        let q_pitch = Quat::from_axis_angle(Vec3::X, r_pitch);
109
110        self.camera_orientation = (q_yaw * q_pitch).normalize();
111    }
112    fn calculate_right(&mut self) {
113        self.right = self.front.cross(self.up).normalize();
114    }
115    fn calculate_front(&mut self) {
116        self.front = self.camera_orientation.mul_vec3(Vec3::NEG_Z).normalize();
117    }
118}
